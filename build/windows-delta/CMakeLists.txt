cmake_minimum_required(VERSION 3.20)
project(windows_delta LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(hd "${VENDOR}/hdiff")
set(sz "${VENDOR}/seven/C")
set(zs "${VENDOR}/zstd/lib")
file(GLOB zstd_sources "${zs}/common/*.c" "${zs}/decompress/*.c")
add_executable(windows-delta main.cpp "${hd}/libHDiffPatch/HPatch/patch.c" ${zstd_sources})
foreach(unit Alloc CpuArch LzFind LzFindOpt LzFindMt Lzma2Dec LzmaDec LzmaEnc Lzma2Enc MtCoder MtDec Threads 7zStream)
  target_sources(windows-delta PRIVATE "${sz}/${unit}.c")
endforeach()
target_compile_definitions(windows-delta PRIVATE _HPATCH_IS_USED_MULTITHREAD=0 _CompressPlugin_zstd=1 ZSTD_DISABLE_ASM=1)
target_include_directories(windows-delta PRIVATE "${hd}" "${sz}" "${zs}")
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(windows-delta PRIVATE Threads::Threads)
endif()
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/apply.cjs" DESTINATION .)
install(TARGETS windows-delta RUNTIME DESTINATION .)
install(FILES "${hd}/LICENSE" DESTINATION licenses RENAME HDiffPatch.txt)
install(FILES "${zs}/../LICENSE" DESTINATION licenses RENAME Zstandard.txt)
install(FILES "${sz}/../DOC/License.txt" DESTINATION licenses RENAME 7-Zip.txt)
